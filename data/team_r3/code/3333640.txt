#define ss second 
#define all(a) (a).begin(),(a).end()
#define rall(a) (a).rbegin(),(a).rend()

const int mod = 1e9 + 7;
int dp[105][10005];
void solve() {
    int n, m; cin >> n >> m;
    vector<pair<int, int>> v(m);
    for(auto &[pos, x] : v) cin >> pos >> x;
    v.insert(v.begin(), {1, 0});
    int ans = 1;
    for(int i = 1; i <= m; i++)
    {
        int dlen = v[i].ff - v[i - 1].ff;
        int dval = v[i].ss - v[i - 1].ss;
        if(dval < 0) 
        {
            ans = 0;
            continue;
        }
        ans = ans * dp[dlen][dval] % mod;
    }
    int tail = n - v[m].ff;
    while(tail--) ans = ans * 2 % mod;
    cout << ans << "\n";

}


signed main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    dp[0][0] = 1;
    for(int i = 1; i <= 104; i++)
    {
        for(int j = 0; j <= 10004; j++) dp[i][j] = dp[i - 1][j];
        for(int j = i; j <= 10004; j++)
        {
            dp[i][j] = (dp[i][j] + dp[i - 1][j - i]) % mod;
        }
    }
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
