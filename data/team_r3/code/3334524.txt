const int inf = 4e18;
void solve() {
    int n;
    cin >> n;
    vector<int> a(n + 1), d(n + 1);

    for(int i = 1; i <= n; i++) {
        cin >> a[i];
    }   
    for(int i = 1; i < n; i++) {
        cin >> d[i];
    }
    

    auto check = [&](int m) -> int {
        vector<array<int, 2>> seg(n + 1);
        // vector<array<bool, 2>> can(n + 1, {0, 0});

        for(int i = 1; i <= n; i++) {
            seg[i][0] = a[i] - m;
            seg[i][1] = a[i] + m;
        }

        // for(int i = 1; i <= n; i++) {
        //     cout << seg[i][0] << " " << seg[i][1] << "\n";
        // }

        // can[1][0] = can[1][1] = 1;

        

        for(int i = 1; i < n; i++) {
            // if(can[i][0]) {
            //     if(llabs(seg[i][0] - seg[i + 1][0]) >= d[i]) can[i + 1][0] = 1;
            //     if(llabs(seg[i][0] - seg[i + 1][1]) >= d[i]) can[i + 1][1] = 1;
            // }

            // if(can[i][1]) {
            //     if(llabs(seg[i][1] - seg[i + 1][0]) >= d[i]) can[i + 1][0] = 1;
            //     if(llabs(seg[i][1] - seg[i + 1][1]) >= d[i]) can[i + 1][1] = 1;
            // }
            int mx = -inf, mn = inf;
            if(seg[i][0] + d[i] <= seg[i + 1][1]) {
                mx = max(mx, seg[i + 1][1]);
                mn = min(mn, max(seg[i + 1][0], seg[i][0] + d[i]));
            }

            if(seg[i][1] - d[i] >= seg[i + 1][0]) {
                mx = max(mx, min(seg[i][1] - d[i], seg[i + 1][1]));
                mn = min(mn, seg[i + 1][0]);
            }

            if(mn == inf && mx == -inf) return 0;
            seg[i + 1][0] = mn;
            seg[i + 1][1] = mx;
        }

        return 1;
    };

    int l = 0, r = 1e18, ans = -1;

    while(l <= r) {
        int mid = (l + r) / 2;

        if(check(mid)) {
            ans = mid;
            r = mid - 1;
        } else l = mid + 1;
    }

    // cout << check(6) << " ";

    cout << ans << "\n";    
}


signed main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
