#include<bits/stdc++.h>
using namespace std;
#define int long long

void solve()
{
    int n, T; cin >> n >> T;
    int mx = 0;
    for(int i = 0, x; i < n; i++)
    {
        cin >> x;
        mx = max(mx, (x + T - 1) / T);
    }
    cout << mx << "\n";
}
signed main()
{
    ios::sync_with_stdio(0); cin.tie(0);
    //int t; cin >> t; while(t--) 
    solve();
}
