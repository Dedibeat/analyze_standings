int f(int x)
{
    if(x == 0) return 1;
    if(x % 3 == 0) return f(x / 3) + 1;
    else return f(x - 1) + 1;
}
int comp(int l, int r)
{
    //hha(l); hhal(r);
    l = max(l, 1LL);
    int ret;
    if(l > r) return 0;
    if(l == r) return f(l);
    else if(r % 3 == 2) 
    {
        ret = comp(l / 3, r / 3) + 3;
    }
    else 
        ret = max(comp(l, r - 1), f(r));
    // hha(l); hha(r);
    // hhal(ret);
    return ret;
    
}
void test(int num, int n)
{
    while(num--)
    {
        int l = rand() % n + 1;
        int r = rand() % n + l;
        int mx = 0;
        for(int i = l; i <= r; i++) mx = max(mx, f(i));
        int ret = comp(l, r);
        if(mx != ret)
        {
            hha(l); hhal(r); hha(mx);
            hhal(ret);
            exit(1);
        }
    }
}
