#define pb push_back

const int N = 3005, inf = 4e18;
int a[N], b[N];
vector<int> dpcur, dpnxt;
int n, m, x, y, z;
int cost(int l, int r, int i)
{
    int tot = llabs(a[r % n] - b[i]) * y;
    assert(l < r);
    if(r >= n) tot += x;
    tot += (r - l - 1) * z;
    return tot;
}
void comp(int l, int r, int lo, int hi, int i)
{
    if(l > r) return;
    int mid = (l + r) / 2;
    pair<int, int> best = {inf, -1};
    for(int k = max(lo, 0LL); k <= min(hi, mid - 1); k++)
    {
        best = min(best, pair<int, int>{dpcur[k] + cost(k, mid, i), k});
    }
    dpnxt[mid] = best.first;
    int opt = best.second;
    comp(l, mid - 1, lo, opt, i);
    comp(mid + 1, r, opt, hi, i);
}
void solve() {
    cin >> n >> m >> x >> y >> z;
    for(int i = 0; i < n; i++) cin >> a[i];
    for(int j = 0; j < m; j++) cin >> b[j];
    dpcur.assign(2 * n, 0);
    for(int i = 0; i < n; i++)
    {
        dpcur[i] = llabs(a[i] - b[0]) * y + i * z + x;
    }
    for(int i = 1; i < m; i++)
    {
        dpnxt.assign(2 * n, inf);
        for(int i = 0; i < n; i++) dpcur[i + n] = dpcur[i];
        comp(0, 2 * n - 1, 0, 2 * n - 1, i);
        dpcur = dpnxt;
        for(int i = 0; i < n; i++) dpcur[i] = min(dpcur[i], dpcur[i + n]);
        //print(dpcur);
    }
    int mn = inf;
    for(int i = 0; i < n; i++)
    {
        mn = min(mn, dpcur[i] + (n - i - 1) * z);
    }
    cout << mn << "\n";
}
int costl(int l, int r, int i)
{
    int tot = llabs(a[r] - b[i]) * y;
    if(r <= l) r += n, tot += x;
    tot += (r - l - 1) * z;
    return tot;
}
int brute()
{
    dpcur.assign(n, 0);
    for(int i = 0; i < n; i++)
    {
        dpcur[i] = llabs(a[i] - b[0]) * y + i * z + x;
    }
    for(int i = 1; i < m; i++)
    {
        dpnxt.assign(n, inf);
        for(int l = 0; l < n; l++)
        {
            for(int r = 0; r < n; r++)
            {
                dpnxt[r] = min(dpnxt[r], dpcur[l] + costl(l, r, i));
            }
        }
        dpcur = dpnxt;
        //print(dpcur);
    }
    int mn = inf;
    for(int i = 0; i < n; i++)
    {
        mn = min(mn, dpcur[i] + (n - i - 1) * z);
    }
    return mn;
}
signed main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
