int f(int x)
{
    if(x == 0) return 1;
    if(x % 3 == 0) return f(x / 3) + 1;
    else return f(x - 1) + 1;
}
int comp(int l, int r)
{
    //hha(l); hhal(r);
    int ret;
    if(l > r) return 0;
    if(r == 0) return 0;
    else if(r % 3 == 2) 
    {
        ret = comp((l + 1) / 3, r / 3) + 3;
    }
    else 
        ret = max(comp(l, r - 1), f(r));
    // hha(l); hha(r);
    // hhal(ret);
    return ret;
    
}
void solve() {
    int l, r; cin >> l >> r;
    cout << comp(l, r) << "\n";

}

    solve();
