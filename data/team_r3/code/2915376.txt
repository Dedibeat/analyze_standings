void solve() {
    int s; cin >> s;
    int f[8];
    FOR(i, 1, 8) cin >> f[i];
    int ans = s / 3;
    ans = min(ans, f[1] + f[3] + f[5] + f[7]);
    ans = min(ans, f[2] + f[3] + f[6] + f[7]);
    ans = min(ans, f[4] + f[5] + f[6] + f[7]);
    ans = min(ans, (s - f[1]) / 2);
    ans = min(ans, (s - f[2]) / 2);
    ans = min(ans, (s - f[4]) / 2);
    cout << ans << '\n';
}   

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t = 1;
    cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}
