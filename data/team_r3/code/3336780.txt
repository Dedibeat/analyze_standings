#define hhal(x) cout << #x << " = " << x << "\n";


void solve() {
    int n; cin >> n;
    deque<int> dq;
    dq.push_back(1);
    for(int i = 2; i <= n; i++) {
        cout << "? " << dq.back() << ' ' << i << endl;
        int x; cin >> x;
        if(x == 1) {
            dq.push_front(i);
        }
        else {
            dq.push_back(i);
        }
    }
    deque<int> ans;
    ans.push_back(dq[0]);
    for(int i = 1; i < n - 1; i++) {
        cout <<  "? " << ans.back() << ' ' << dq[i] << endl;
        int x; cin >> x;
        if(x == 1) {
            ans.push_front(dq[i]);
        }
        else {
            ans.push_back(dq[i]);
        }
    }
    ans.push_back(dq.back());
    cout << "! ";
    for(auto &x : ans) {
        cout << x << ' ';
    }
    cout << endl;
}


signed main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
