#define hhal(x) cout << #x << ": " << x << endl;

void solve()
{
    int n, m; cin >> n >> m;
    vector<vector<int>> g(n);
    vector<int> d(n);
    for(auto &x : d) cin >> x;
    for(int i = 0; i < n; i++)
    {
        int sz; cin >> sz;
        g[i].assign(sz, 0);
        for(auto &v : g[i]) cin >> v, v--;
        sort(all(g[i]), [&](int u, int v)
        {
            return d[u] == d[i] + 1;
        });
    }
    vector<int> vis(n), dd(n);
    auto dfs = [&](auto &&self, int u) -> void
    {
        vis[u] = true;
        for(int v : g[u])
        {
            if(vis[v]) continue;
            dd[v] = dd[u] + 1;
            self(self, v);
        }
    };
    int root = -1;
    for(int i = 0; i < n; i++)
    {
        if(d[i] == 0) 
        {
            if(root == -1)
            {
                root = i;
            }
            else 
            {
                cout << "-1\n";
                return;
            }
        }
    }
    if(root == -1)
    {
        cout << "-1\n";
        return;
    }
    dfs(dfs, root);
    if(dd != d)
    {
        cout << "-1\n";
        return;
    }
    cout << root + 1 << "\n";
    for(int i = 0; i < n; i++)
    {
        for(auto x : g[i]) cout << x + 1 << " ";
        cout << "\n";
    }
}
signed main()
{
    ios::sync_with_stdio(0); cin.tie(0); 
    // int t; cin >> t; while(t--)
    solve();
}
