int check(int L, int R, vector<pair<int, int>> v)
{
    int cur = L, sing = 1;
    for(auto [l, r] : v)
    {
        if(sing == 0) continue;
        if(l + 1 == r)
        {
            if(cur == l) 
            {
                cur = l + 1;
                sing = 1;
            }
            else if(cur == l + 1)
            {
                cur = l;
                sing = 1;
            }
            else 
            {
                sing = 0;
            }
        }
    }
    if(sing && cur == R) return 0;
    else return 1;
}
void solve() {
    int n; cin >> n;
    vector<pair<int, int>> v(n);
    for(int i = 0; i < n; i++) cin >> v[i].first;
    for(int i = 0; i < n; i++) cin >> v[i].second;
    auto Sing = [&](int i)
    {
        if( v[i].first == v[i].second ) return v[i].first;
        else return 0LL;
    };
    vector<array<int, 4>> seg;
    int beg = 0;
    //for(int i = 0; i < n; i++) hhal(Sing(i));
    for(int i = 1; i < n; i++)
    {
        if(Sing(i - 1) != Sing(i))
        {
            seg.push_back({beg, i - 1, 0, i - beg});
            beg = i;
        }
    }
    seg.push_back({beg, n - 1, 0, n - beg});
    for(auto &[l, r, s, len] : seg)
    {
        if(Sing(l)) 
        {
            s = len * (len - 1) / 2;
        }
        //hha(l); hha(r); hha(s); hhal(len);
    }
    for(int i = 1; i + 1 < seg.size(); i++)
    {
        auto &[l, r, s, len] = seg[i];
        if(Sing(l)) continue;
        vector<pair<int, int>> cur(v.begin() + l, v.begin() + r + 1);
        int L = Sing(seg[i - 1][0]);
        int R = Sing(seg[i + 1][0]);
        //hha(L); hhal(R);
        int ok = check(L, R, cur);
        //hhal(ok);
        if(cur.size() >= 2)
        {
            s = ok;
        }
        else if(!ok)
        {
            if(seg[i - 1][3] <= seg[i + 1][3])
            {
                seg[i - 1][2] += seg[i - 1][3];
                seg[i - 1][3]++;
            }
            else 
            {
                seg[i + 1][2] += seg[i + 1][3];
                seg[i + 1][3]++;
            }
        }
    }
    int ans = 0;
    for(auto &[l, r, s, len] : seg) 
    {
        //hha(l); hha(r); hha(s); hhal(len);
        ans += s;
    }
    //hhal(ans);
    cout << ans * 2 + n << "\n";
}
signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
}
