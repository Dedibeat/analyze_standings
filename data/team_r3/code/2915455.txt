int Pow[200005];
const int mod = 998244353;
void solve() {
    int n; cin >> n;
    vector<int> a(n);
    for(auto &x : a) cin >> x;
    sort(all(a));
    int ans = a[0] * Pow[n - 1] % mod;
    //print(a);
    int x = 0;
    for(int i = 1; i < n; i++)
    {
        x = (x * 3 + 1) % mod;
        int comb = (x + 1) * Pow[n - i - 1] % mod;
        //hha(i); hhal(comb);
        ans = (ans + comb * a[i]) % mod;
    }
    // int tot = 0;

    // }
    // cout << tot % mod << endl;
    cout << ans << "\n";
}   

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    Pow[0] = 1;
    for(int i = 1; i < 2e5 + 5; i++) Pow[i] = Pow[i - 1] * 2 % mod;
    int t = 1;
    cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}
