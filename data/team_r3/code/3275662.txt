const int N = 4e5 + 5;
int f[N];
int get(int u)
{
    return f[u] == u ? u : (f[u] = get(f[u]));
}
void link(int u, int v)
{
    u = get(u); v = get(v);
    if(u != v) f[v] = u;
}
int solve(vector<int> a, int n) {
    iota(f, f + 2 * n + 1, 0);
    vector<int> prv(n + 1);
    for(int i = 1; i <= 2 * n; i++)
    {
        if(prv[a[i]] == 0)
        {
            prv[a[i]] = i;
        }
        else 
        {
            link(prv[a[i]], i);
        }
    }
    string s = "(";
    vector<int> stk;
    stk.push_back(1);
    int cnt = 0;
    for(int i = 2; i < 2 * n; i++)
    {
        if(stk.empty())
        {
            stk.push_back(i);
            s += '(';
        }
        else 
        {
            int tp = stk.back();
            int ok = (get(i) != get(tp));
            if(!ok)
            {
                stk.push_back(i);
                s += '(';
            }
            else 
            {
                stk.pop_back();
                s += ')';
                link(tp, i);
                cnt++;
            }
        }
    }
    if(stk.size() != 1 || cnt != n - 1) 
    {
        //print(stk);
        return 1;
    }
    s += ')';
    cout << s << "\n";
    return 0;
    
}
void test(int n, int num)
{
    mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
    while(num--)
    {
        vector<int> a(2 * n + 1);
        vector<int> b(2 * n);
        for(int i = 0; i < n; i++) b[i] = b[i + n] = (i + 1);
        
        shuffle(all(b), rng);
        //print(b);
        for(int i = 1; i <= 2 * n; i++)
        {
            a[i] = b[i - 1];
        }
        if(solve(a, n))

        }
    }

}
