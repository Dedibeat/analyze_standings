int f(int x, int y)
{
    if(x < y) return y - x;
    int mn = x - y;
    int cnt = 0;
    while(x > 1)
    {
        x /= 2;
        cnt++;
        mn = min(mn, cnt + llabs(x - y));
    }
    return mn;
}
void solve() {
    int n, m; cin >> n >> m;
    m = n - m;
    vector<int> a(n);
    for(int &x : a) cin >> x;
    auto comp = [&](int A)
    {
        vector<int> b;
        for(int x : a)
        {
            b.push_back(f(x, A));
        }
        sort(all(b));
        //print(b);
        int sum = 0;
        for(int i = 0; i < m; i++) sum += b[i];
        return sum;
    };
    // for(int i = 1; i <= 30; i++) 
    // {
    //     cout << comp(i) << " ";
    // }
    // cout << endl;
    int lo = 1, hi = 2e9;
    while(lo + 1 < hi)
    {
        int mid = (lo + hi) / 2;
        if(comp(mid) > comp(mid + 1))
        {
            lo = mid;
        }
        else hi = mid;
    }
    cout << comp(lo + 1) << "\n";
}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    srand(time(NULL));
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
