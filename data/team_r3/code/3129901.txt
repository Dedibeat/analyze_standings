void solve() {
    int n, k; cin >> n >> k;
    int x = (n % k);
    if(x == 0) x = k;
    int y = k - x;
    int t = n / 2;
    int a = t, b = -t;
    int A = (n + k - 1) / k, B = A - 1;
    {
        int d = (a / B);
        a -= d * B;
        b += d * A;
    }
    if(a > x || b < 0)
    {
        cout << "NO\n";
        return;
    }
    if(b > y)
    {
        int dif = (b - y);
        int d = (dif + A - 1) / A;
        b -= d * A;
        a += d * B;
    }
    if(a > x || b < 0)
    {
        cout << "NO\n";
        return;
    }
    //hhal(n); hha(a); hha(b); hhal(a * A + b * B);
    cout << "YES\n";
}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    srand(time(NULL));
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
