#define pb push_back
#define all(x) x.begin(), x.end()
#define rall(x) x.rbegin(), x.rend()
#define FOR(i, a, b) for(int i = (a); i < (b); i++)
#define FORD(i, a, b) fosr(int i = (a); i >= (b); i++)

void solve() {
    int n, k; cin >> n >> k;
    int mn = LLONG_MAX;
    FOR(i, 1, k + 1) {
        int x; cin >> x;
        mn = min(x, mn);
    }
    map<string, vector<int>> mp;
    vector<int> b(n + 1);
    FOR(i, 1, n + 1) {
        int a; cin >> a;
        b[i] = a;
        string s; cin >> s;
        mp[s].pb(a);
    }
    vector<int> v;
    for(auto [s, a] : mp) {
        sort(rall(a));
        FOR(i, 0, min((ll)a.size(), mn)) {
            v.pb(a[i]);
        }
    }
    sort(all(v));
    FOR(i, 1, n + 1) {
        int pos = lower_bound(all(v), b[i]) - v.begin();
        cout << v.size() - pos << '\n';
    }
}   

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t = 1;
    //cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}
