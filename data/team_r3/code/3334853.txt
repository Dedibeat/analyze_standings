const int mod = 998244353;
int dp[105][105][10005];
int solve(int n, int m, vector<pair<int, int>> v) {
    v.insert(v.begin(), {1, 0});
    int ans = 1;
    vector<int> dpcur(n + 5, 0), dpnxt;
    dpcur[0] = 1;
    int lim = n * (n - 1) / 2;
    for(int i = 0; i < m; i++)
    {
        dpnxt.assign(n + 5, 0);
        int dlen = v[i + 1].ff - v[i].ff;
        int dval = v[i + 1].ss - v[i].ss;
        for(int j = 0; j <= v[i].ff; j++)
        {
            for(int dj = 0; dj <= dlen; dj++)
            {
                int k = j + dj;
                int dif = dval - j * dlen;
                if(dif < 0 || dif > lim) continue;
                dpnxt[k] = (dpnxt[k] + dpcur[j] * dp[dlen][dj][dif] % mod) % mod;
            }
        }
        dpcur = dpnxt;
    }
    ans = accumulate(all(dpcur), 0LL) % mod;
    int tail = n - v[m].ff;
    while(tail--) ans = ans * 2 % mod;
    return ans;

}
int brute(int n, int m, vector<pair<int, int>> v)
{
    
    int cnt = 0;
    for(int mask = 0; mask < (1 << (n - 1)); mask++)
    {
        vector<int> a(n);
        for(int i = 0; i < n - 1; i++)
        {
            if(mask & (1 << i)) a[i + 1] = 1;
            //a[i + 1] += a[i];
        }
        //print(a);
        
        for(int i = 1; i < n; i++) a[i] += a[i - 1];
        auto s = a;
        for(int i = 1; i < n; i++) a[i] += a[i - 1];
        int ok = 1;
        for(auto [pos, x] : v) 
        {
            if(a[pos - 1] != x) ok = 0;
        }
        //hhal(mask);
        //print(a); 
        //print(s);
        cnt += ok;
    }
    return cnt;
}
void test(int trial, int den, int den2, int n)
{
    while(trial--)
    {
        vector<int> a(n), vis(n);
        for(int i = 1; i < n; i++)
        {
            if(rand() % den == 0) a[i] = 1;
            if(rand() % den2 == 0) vis[i] = 1;
        }
        auto o = a;
        vis[0] = (rand() % den2 == 0);
        for(int i = 1; i < n; i++) a[i] += a[i - 1];
        for(int i = 1; i < n; i++) a[i] += a[i - 1];
        vector<pair<int, int>> v;
        for(int i = 0; i < n; i++)
        {
            if(vis[i]) v.push_back({i + 1, a[i]});
        }
        int m = v.size();
        int ret1 = solve(n, m, v);
        int ret2 = brute(n, m, v);
        if(ret1 != ret2)
        {
            hha(trial); hha(ret1); hhal(ret2);
            print(o);
            print(a);
            print(vis);
            exit(0);
        }
        hhal(ret1);
    }
    exit(0);
}
