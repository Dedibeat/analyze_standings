int comp(string s, string t, int flg)
{
    int n = s.size(), m = t.size();
    vector<vector<int>> dp(n, vector<int>(n));
    vector<vector<int>> cnt(n, vector<int>(n + 1));
    for(int i = 0; i < n; i++)
    for(int j = i + 1; j < n; j++)
    {
        if(s[i] == s[j])
        {
            dp[i][j] = (i ? dp[i - 1][j - 1] : 0) + 1;
        }
        if(i + 2 <= j)
        {
            int k = max(i + 2, j + 1 - dp[i][j]);
            //log(i); log(j); logl(k);
            cnt[i + 1][k - 1]++;
            cnt[i + 1][j]--;
        }
    }
    vector<vector<int>> dp2(n + 1, vector<int>(m + 1));
    vector<vector<int>> cnt2(n + 1, vector<int>(n + 1));

    for(int i = n - 1; i >= 0; i--)
    for(int j = m - 1; j >= 0; j--)
    {
        if(s[i] == t[j]) dp2[i][j] = dp2[i + 1][j + 1] + 1;
        int k = i + dp2[i][j];
        cnt2[i][i]++;
        cnt2[i][k]--;
    }
    int ans = 0;
    for(int i = 0; i < n; i++)
    {
        int sum = flg, need = 0;
        for(int j = i; j < n; j++)
        {
            sum += cnt[i][j];
            need += cnt2[i][j];
            ans += sum * need;
        }
    }
    //logl(ans);
    return ans;
}
void solve()
{
    string s, t; cin >> s >> t;
    int ans = 0;
    ans += comp(s, t, 1);
    ans += comp(t, s, 0);
    cout << ans << endl;
}
signed main()
{
    ios::sync_with_stdio(0); cin.tie(NULL);
    solve();
}
