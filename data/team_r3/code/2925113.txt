const int mod = 998244353;

int modpow(int a, int b) {
    a %= mod;
    int c = 1;
    while(b) {
        if(b & 1) c = c * a % mod;
        a = a * a % mod;
        b /= 2;
    }
    return c;
}

int modinv(int a) {
    return modpow(a, mod - 2);
}
void solve() {
    int x, y; cin >> x >> y;
    int a, b, c; cin >> a >> b >> c;
    c = (a + b);
    a = a * modinv(c) % mod;
    b = b * modinv(c) % mod;
    //hha(a); hhal(b);
    int ans = 0;
    auto g = [&](int a, int b, int k)
    {
        if(k == 0) return 0LL;
        if(b == 1) return a * k % mod;
        int aa = a * (modpow(b, k) - 1 + mod) % mod;
        int bb = (b - 1 + mod) % mod;
        return aa * modinv(bb) % mod;
    };
    auto f = [&](auto &&self, int x, int y, int p = 1) -> void
    {
        //hha(x); hha(y); hhal(p);
        if(y <= 0) 
        {
            ans = (ans + p) % mod;
            return;
        }
        if(x <= 0) return;
        if(x > y)
        {
            int k = x / y;
            ans = (ans + p * g(a, b, k)) % mod;
            p = p * modpow(b, k) % mod;
            x = x % y;
            self(self, x, y, p);
        }
        else 
        {
            int k = y / x;
            p = p * modpow(a, k) % mod;
            y = y % x;
            self(self, x, y, p);
        }
    };
    f(f, x, y);
    cout << ans << "\n";
} 

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t = 1;
    cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}
