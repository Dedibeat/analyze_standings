void solve()
{
    int n, k; cin >> n >> k;
    vector<int> a(n); for(auto &x : a) cin >> x;
    sort(a.begin(), a.end());
    int c = (a[0] + k) * 2, cur = 0, b = 0;
    for(int i = 0; i < n; i++) {
        cur += a[i];
        if(i) c = min(c, (cur + k - 1) / (i));
    }
    for(int i = 0; i < n; i++) b += max(0ll, c - a[i]);
    int ans = c * (n - 1) + 1, mn = a[0];
    for(int i = n - 1; i >= 0; i--) {
        mn = min(mn, a[i] - 1);
        if(a[i] <= c) b++;
        if(b <= c - 1 + k) ans++;
        else break;
    }
    ans = min((c + 1) * (n - 1), ans);
    cout << ans << '\n';
}
signed main()
{
    ios::sync_with_stdio(0); cin.tie(0); 
    int t; cin >> t; while(t--)
    solve();
}
