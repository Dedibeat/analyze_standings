#define rep(i, a, b) for(int i = (a); i < (b); i++)
const int inf = 1e18;
void solve()
{
    int n, m, cap, t; cin >> n >> m >> cap >> t;
    vector<vector<pair<int, int>>> g(n + 1);
    while(m--)
    {
        int u, v, w; cin >> u >> v >> w;
        g[u].push_back({w, v});
        g[v].push_back({w, u});
    }
    using nd = array<int, 3>;
    priority_queue<nd, vector<nd>, greater<>> q;
    vector<pair<int, int>> d(n + 1, {inf, 0});
    d[t] = {1, 0};
    q.push({1, 0, t});
    vector<int> vis(n + 1);
    while(q.size())
    {
        auto [c, used, u] = q.top(); q.pop();
        if(vis[u]) continue;
        vis[u] = true;
        for(auto [w, v] : g[u])
        {
            int c1 = c, used1 = used;
            if(used1 + w > cap) c1++, used1 = w;
            else used1 += w;
            if(pair<int, int>{c1, used1} < d[v])
            {
                d[v] = {c1, used1};
                q.push({c1, used1, v});
            }
        }
    }
    for(int i = 1; i <= n; i++)
    {
        if(d[i].first >= inf) cout << "-1 ";
        else cout << d[i].first << " ";
    }
    cout << "\n";
}
signed main()
{
    ios::sync_with_stdio(0); cin.tie(NULL);
    //int t; cin >> t; while(t--)
    solve();
}
