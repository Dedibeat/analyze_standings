int egcd(int a, int b, int &x, int &y)
{
    if(!b) 
    {
        x = 1; y = 0;
        return a;
    }
    int x1, y1;
    int g = egcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - y1 * (a / b);
    return g;
}
pair<int, int> brute(int S, int A, int B, int m)
{
    int mn = m - 1, s = 0, d = 0;
    for(int i = 0; i < m; i++)
    for(int j = 0; j < m; j++)
    {
        int tot = (S + A * i + B * j) % m;
        if(tot < mn) mn = tot, d = i, s = j;
    }
    return {s, d};
}
pair<int, int> solve(int S, int A, int B, int m) {

    int s = 0, d = 0;
    //hha(A); hhal(B);
    int g = egcd(A, B, d, s);
    d = (d % m + m) % m;
    s = (s % m + m) % m;
    //hha(d); hha(s); hhal(g);
    int invg = 0, tmp = 0;
    int gg = egcd(g, m, invg, tmp);
    invg = (invg % m + m) % m;
    int mn = S % gg;
    int R = (mn - S + m) % m;
    int t = (R / gg) * invg % m;
   // hhal(mn);
   // hhal(t);
    d = d * t % m;
    s = s * t % m;
    return {s, d};
}
void test(int num, int m)
{
    while(num--)
    {
        int S = rand() % m;
        int A = rand() % m;
        int B = rand() % m;
        auto [s1, d1] = brute(S, A, B, m);
        auto [s, d] = solve(S, A, B, m);
        int mn1 = (S + A * d1 + B * s1) % m;
        int mn = (S + A * d % m + B * s % m) % m;
        assert(mn1 >= 0 && mn >= 0);
        if(mn != mn1) 
        {
            hha(S); hha(A); hha(B); hhal(m);
            hha(s); hhal(d);
            hha(mn); hhal(mn1);
            exit(1);
        }
    }
}
