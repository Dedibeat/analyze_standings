template<typename T, typename U>
ostream &operator<<(ostream &os, const pair<T, U> &p)
{
    return os << "(" << p.F << "," << p.S << ")";
}
using Frac = pair<int, int>;
struct compFrac
{
    bool operator()(Frac x, Frac y) const 
    {
        if(x.S < 0) x.F = -x.F, x.S = -x.S;
        if(y.S < 0) y.F = -y.F, y.S = -y.S;
        return x.F * y.S < y.F * x.S;
    }
};

int comp(vector<int> a, vector<int> b, int n)
{
    int ans = 0;
    map<Frac, array<int, 2>, compFrac> mp;
    int cnt = 0;
    for(int i = 0; i < n; i++)
    {
        if(a[i] == 0 || b[i] == 0)
        {
            ans += ((a[i] + b[i]) < 0 ? 1 : 0);
            continue;
        }
        if(a[i] * b[i] > 0)
        {
            if(b[i] < 0) ans++;
            continue;
        }
        if(b[i] > 0) mp[{a[i], -b[i]}][0]++, cnt++;
        if(b[i] < 0) mp[{a[i], -b[i]}][1]++;
    }
    int mx = cnt;
    //logl(cnt);
    for(auto [f, s] : mp)
    {
        cnt -= s[0];
        //log(f); log(s[0]); log(s[1]); logl(cnt);
        cnt += s[1];
        mx = max(mx, cnt);
    }
    return ans + mx;

}
void flip(vector<int> &v) 
{
    for(auto &x : v) x = -x;
}
void solve()
{
    int n; cin >> n;
    vector<int> a(n), b(n);
    for(int i = 0; i < n; i++)
    {
        int _; cin >> a[i] >> b[i] >> _;
    }
    int ans = 0;
    ans = max(ans, comp(a, b, n));
    flip(a);
    ans = max(ans, comp(a, b, n));
    flip(b);
    ans = max(ans, comp(a, b, n));
    flip(a);
    ans = max(ans, comp(a, b, n));
    cout << ans << "\n";
}
signed main()
{
    ios::sync_with_stdio(0); cin.tie(NULL);
    int T; cin >> T; while(T--) 
    solve();
}
