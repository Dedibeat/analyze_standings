const int N = 1e6 + 1;
const int inf = 1e18;
int a[N];
int dp[N][4];
vector<int> g[N];
void dfs(int u, int p = -1)
{
    vector<int> ch;
    for(int v : g[u])
    {
        if(v == p) continue;
        dfs(v, u);
        ch.push_back(v);
    }
    int s = -inf, t = -inf, indef = 0;
    for(int v : ch)
    {
        indef += dp[v][3];
        if(a[v] < a[u])
        {
            s = max(s, -a[v] + dp[v][0] - dp[v][3]);
        }
        if(a[v] > a[u])
        {
            t = max(t, a[v] + dp[v][1] - dp[v][3]);
        }
    } 
    dp[u][0] = max(indef, indef + s + a[u]);
    dp[u][1] = max(indef, indef + t - a[u]);
    dp[u][2] = max(indef, indef + s + t);
    dp[u][3] = max({dp[u][0], dp[u][1], dp[u][2]});
}
void solve()
{
    int n; cin >> n;
    for(int i = 1; i <= n; i++) 
        cin >> a[i];
    for(int i = 1; i < n; i++)
    {
        int u, v; cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1);
    cout << dp[1][3] << "\n";
}
signed main()
{
    ios::sync_with_stdio(0); cin.tie(NULL);
    //int t; cin >> t; while(t--)
    solve();
}
