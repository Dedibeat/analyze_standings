void solve() {
    int n; cin >> n;
    unordered_set<string> st;
    auto check = [&](string &s)
    {
        for(int i = 2; i < s.size(); i++)
        {
            if(s[i - 2] == 'b' && s[i - 1] == 'i' && s[i] == 'e')
                return true;
        }
        return false;
    };
    for(int i = 0; i < n; i++)
    {
        int m; cin >> m;
        int ok = 0;
        while(m--)
        {
            string s; cin >> s;
            if(check(s) && !st.count(s)) 
            {
                cout << s << "\n";
                st.insert(s);
                ok = 1;
            }
        }
        if(!ok) cout << "Time to play Genshin Impact, Teacher Rice!\n";
    }
}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    solve();
    return 0;
}
