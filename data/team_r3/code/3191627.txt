#define int long long
#define pb push_back
#define all(x) (x).begin(), (x).end()
#define rall(x) (x).rbegin(), (x).rend()
#define rep(i, a, b) for(int i = (a); i < (b); i++)
#define repd(i, a, b) for(int i = (a); i >= (b); i--)

const int mod = 1e9 + 7;
void solve() {

    int a, b, c;
    cin >> a >> b >> c;

    int p, q, r;
    cin >> p >> q >> r;

    int n;
    cin >> n;
    vector<int> x(n), y(n), z(n);

    for(int i = 0; i < n; i++) {
        cin >> x[i] >> y[i] >> z[i];
    }

    sort(all(x));
    sort(all(y));
    sort(all(z));

    if(p >= a || q >= b || r >= c) {
        cout << "0\n";
        return;
    }
    // cout << n << " " << (p + 1) * (q + 1) * (r + 1) << "\n";
    if(n % ((p + 1) * (q + 1) * (r + 1)) != 0) {
        cout << "0\n";
        return;
    }


    int ans = 1;
    int len = n / (1 + p);
    for(int i = 1; i <= p; i++) {
        ans = ans * (x[i * len] - x[i * len - 1]) % mod;
    }
    len = n / (1 + q);
    for(int i = 1; i <= q; i++) {
        ans = ans * (y[i * len] - y[i * len - 1]) % mod;
    }
    len = n / (1 + r);
    for(int i = 1; i <= r; i++) {
        ans = ans * (z[i * len] - z[i * len - 1]) % mod;
    }
    cout << ans << "\n";
}
signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    //int t; cin >> t; while(t--)
    solve();
}
