void solve() {
    int n; cin >> n;
    auto get = [&](int L)
    {
        if(L == 0) return pair<int, int>{1e18, 1};
        int R = (n - L);
        int a = L * (L + 1) / 2 + R;
        int b = L;
        //int g = __gcd((uint)a, (uint)b);
        return pair<int, int>{a, b};
    };
    auto check = [&](int L)
    {
        auto [a1, b1] = get(L - 1);
        auto [a2, b2] = get(L);
        return (bint)a1 * b2 >= (bint)a2 * b1;
    };
    int lo = 1, hi = n + 1;
    while(lo + 1 < hi)
    {
        int mid = (lo + hi) / 2;
        if(check(mid)) lo = mid;
        else hi = mid;
    }
    auto [a, b] = get(lo);
    int g = __gcd((uint)a, (uint)b);
    cout << a / g << " " << b / g << "\n";
}   

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t = 1;
    cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}
