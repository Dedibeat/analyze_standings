const uint64_t HMOD = (1ULL << 61) - 1;
const uint64_t HB =
    mt19937_64(chrono::steady_clock::now().time_since_epoch().count())()
    % (HMOD - 300) + 257;

struct Hash {
    vector<uint64_t> h, p;

    static uint64_t mul(uint64_t a, uint64_t b) {
        __uint128_t c = (__uint128_t)a * b;
        uint64_t r = (uint64_t)(c >> 61) + ((uint64_t)c & HMOD);
        return r >= HMOD ? r - HMOD : r;
    }

    Hash(const string& s) {
        int n = s.size();
        h.assign(n + 1, 0);
        p.assign(n + 1, 1);
        for (int i = 0; i < n; i++) {
            h[i + 1] = mul(h[i], HB) + (unsigned char)s[i] + 1;
            if (h[i + 1] >= HMOD) h[i + 1] -= HMOD;
            p[i + 1] = mul(p[i], HB);
        }
    }

    // hash of s[l..r)
    uint64_t get(int l, int r) {
        uint64_t x = h[r] + HMOD - mul(h[l], p[r - l]);
        return x >= HMOD ? x - HMOD : x;
    }
};
int comp(string s, string t, int flg)
{
    int n = s.size(), m = t.size();
    vector<vector<int>> dp(n, vector<int>(n));
    vector<vector<int>> cnt(n, vector<int>(n + 1));
    for(int i = 0; i < n; i++)
    for(int j = i + 1; j < n; j++)
    {
        if(s[i] == s[j])
        {
            dp[i][j] = (i ? dp[i - 1][j - 1] : 0) + 1;
        }
        if(i + 2 <= j)
        {
            int k = max(i + 2, j + 1 - dp[i][j]);
            //log(i); log(j); logl(k);
            cnt[i + 1][k - 1]++;
            cnt[i + 1][j]--;
        }
    }
    Hash hs(s), ht(t);
    map<int, int> mp;
    for(int i = 0; i < n; i++)
    {
        int sum = flg;
        for(int j = i; j < n; j++)
        {
            sum += cnt[i][j];
            //log(i); log(j); logl(sum);     
            mp[hs.get(i, j + 1)] += sum;       
            cnt[i][j] = sum;
        }
    }

    int ans = 0;
    for(int i = 0; i < m; i++)
    for(int j = i; j < m; j++)
    {
        ans += mp[ht.get(i, j + 1)];
    }
    //logl(ans);
    return ans;
    
}
void solve()
{
    string s, t; cin >> s >> t;
    int ans = 0;
    ans += comp(s, t, 1);
    ans += comp(t, s, 0);
    cout << ans << "\n";
}

    ios::sync_with_stdio(0); cin.tie(NULL);
    solve();
