#define rep(i, a, b) for(int i = (a); i < (b); i++)
#define repd(i, a, b) for(int i = (a); i >= (b); i--)

void solve() {
    int n;
    cin >> n;
    vector<int> a(n + 10, 0);
    int cnt = 0;
    for(int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        if(a[x + 1]) {
            a[x + 1] = 0;
            a[x] = 1;
        } else {
            cnt++;
            a[x] = 1;
        }
    }
    cout << cnt << "\n";
}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    srand(time(NULL));
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
