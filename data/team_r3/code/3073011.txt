void solve() {
    int a, b, c;
    cin >> a >> b >> c;
    int ans = 2e18;
    for(int i = 0; i <= 30; i++) {
        int cur = a * i;

        int cnt = (1 << i);

        cur += ((c + cnt - 1) / cnt) * b;
        ans = min(ans, cur);
        // cout << cur << " ";
    }
    cout << ans << "\n";


}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
