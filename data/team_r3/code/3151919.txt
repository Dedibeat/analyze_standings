#include <bits/stdc++.h>
using namespace std;

#define int long long
#define pb push_back
#define all(x) (x).begin(), (x).end()
#define rall(x) (x).rbegin(), (x).rend()
#define rep(i, a, b) for(int i = (a); i < (b); i++)
#define repd(i, a, b) for(int i = (a); i >= (b); i--)
#define hha(x) cout << #x << ' ' << x << ' ';
#define hhal(x) cout << #x << ' ' << x << '\n';
#define print(v) cout << #v << ' '; for(auto &x : v) cout << x << ' '; cout << '\n';

void solve() {
    int n;
    cin >> n;

    vector<int> a(n + 1), l(n + 1, -1), r(n + 1, -1), b, c, ans(n + 1, -1);

    for(int i = 1; i <= n; i++) {
        cin >> a[i];
        if(l[a[i]] != -1) r[a[i]] = i;
        else l[a[i]] = i;
    }


    for(int i = 1; i <= n; i++) {
        if(l[i] == -1) b.push_back(i);
    }

    int id = 0;

    if(r[a[1]] == -1) {

        for(int i = 1; i <= n; i++) {
            if(r[a[i]] == -1 && r[a[i]]) c.push_back(a[i]);
            else if(r[a[i]] != i) {
                ans[i - 1] = b[id];
                ans[r[a[i]] - 1] = b[id];
                id++; 
            } 
        }


        id = 0;
        for(int i = 1; i <= n; i++) {
            if(ans[i] == -1) ans[i] = c[id++];
        }

        cout << 0 << "\n";
        for(int i = 1; i <= n; i++) cout << ans[i] << " ";
        cout << "\n";

    } else {

        for(int i = 2; i <= n; i++) {
            if(r[a[i]] == -1 && r[a[i]]) c.push_back(a[i]);
            else if(r[a[i]] != i) {
                ans[i - 1] = b[id];
                ans[r[a[i]] - 1] = b[id];
                id++; 
            } 
        }

        int cnt = 0;
        for(int i = 1; i <= n; i++) {
            if(ans[i] == -1) {
                ans[i] = b[id];
                cnt++;
            }
            if(cnt == 2) break;
        }
        id = 0;
        for(int i = 1; i <= n; i++) {
            if(ans[i] == -1) ans[i] = c[id++];
        }

        cout << 1 << "\n";
        for(int i = 1; i <= n; i++) cout << ans[i] << " ";
        cout << "\n";

    }

}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
}
