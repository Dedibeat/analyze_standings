#include <bits/stdc++.h>
using namespace std;

#define int long long
#define ff first
#define ss second
#define pb push_back
#define all(a) (a).begin(),(a).end()
#define rall(a) (a).rbegin(),(a).rend()
#define sz(a) (int)(a).size()

void solve() {
    int a, r = 0;
    cin >> a;
    for(int i = 1; i <= 31; i++) {
        int b;
        cin >> b;
        if(b < a) r++;
    }
    if(r < 2) cout << "32\n";
    else if(r < 6) cout << "16\n";
    else if(r < 13) cout << "8\n";
    else if(r < 27) cout << "4\n";
    else if(r < 31) cout << "2\n";
    else cout << "1\n";
}

signed main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    int t = 1;
    cin >> t;
    while(t--) solve();
    return 0;
}
