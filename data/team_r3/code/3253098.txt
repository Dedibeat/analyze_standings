#include <bits/stdc++.h>
using namespace std;
#define int long long
#define ff first 
#define ss second 
#define all(a) (a).begin(),(a).end()
#define rall(a) (a).rbegin(),(a).rend()
#define print(x) cout << #x << " = "; for(auto i : x) cout << i << " "; cout << "\n";
#define hha(x) cout << #x << " = " << x << " ";
#define hhal(x) cout << #x << " = " << x << "\n";


void solve() {
    int n; cin >> n;
    vector<int> a(n);
    for(int &x : a) cin >> x;
    sort(all(a));
    int g = 0;
    for(int i = 1; i < n; i++) g = gcd(g, a[i] - a[i - 1]);
    if(g == 0)
    {
        cout << "infinite\n";
        return;
    }
    int k = g / gcd(g, a[0]);
    cout << g << " " << k << "\n";

}


signed main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
