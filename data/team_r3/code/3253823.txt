const int inf = 2e18;

int dp[105][105];

void solve() {
    int n, m;
    cin >> n >> m;
    int k;
    cin >> k;

    vector<array<int, 2>> p(k);
    vector<int> pos(m + 1, -1);
    int id = 0;

    for(auto &[r, c] : p) {
        cin >> r >> c;
        pos[c] = id;
        id++;
    }
    // print(pos);

    auto bod = [&](int mask) -> int {
        for(int j = 2; j <= m; j++) {
            for(int i = 1; i <= n; i++) {

                // if(pos[j] != -1) {
                //     cout << j << "\n";
                // }

                if(pos[j] != -1 && (((mask & (1 << pos[j])) && p[pos[j]][0] >= i) || (!(mask & (1 << pos[j])) && p[pos[j]][0] <= i))) {
                    dp[i][j] = inf;
                    // cout << "hahah\n";
                }
                else if(dp[i][j - 1] != inf) dp[i][j] = dp[i][j - 1] + 1;
                else dp[i][j] = inf;
            }

            int val = inf;
            for(int i = 1; i <= n; i++) {
                if(dp[i][j] == inf) {
                    val++;
                    continue;
                }

                if(dp[i][j] < val) val = dp[i][j];
                else dp[i][j] = val;
                val++;
            }
            val = inf;

            for(int i = n; i > 0; i--) {
                if(dp[i][j] == inf) {
                    val++;
                    continue;
                }

                if(dp[i][j] < val) val = dp[i][j];
                else dp[i][j] = val;
                val++;
            }
        } 
        int ans = inf;
        for(int i = 1; i <= n; i++) {
            if(dp[i][m] < inf) ans = min(dp[i][m], ans);
        }
        return (ans >= inf ? -1 : ans);
    };
    for(int i = 0; i < (1 << k); i++) {
        cout << bod(i) << " ";
    }   
    cout << '\n';
}


signed main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
