void solve() {
    int a, b, ha, hb; cin >> a >> b >> ha >> hb;
    int k = (hb + a - 1) / a - 1;
    if(ha - b * (k + 4) > 0) {
        cout << "Yes\n";
    }
    else {
        cout << "No\n";
    }

}

    ios_base::sync_with_stdio(0);cin.tie(0);

    solve();
