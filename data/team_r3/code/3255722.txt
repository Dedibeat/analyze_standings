vector<int> manacher_odd(string s) {
    int n = s.size();
    s = char(1) + s + char(2);
    vector<int> p(n + 2);
    int l = 0, r = 1;
    for(int i = 1; i <= n; i++) {
        p[i] = min(r - i, p[l + (r - i)]);
        while(s[i - p[i]] == s[i + p[i]]) {
            p[i]++;
        }
        if(i + p[i] > r) {
            l = i - p[i], r = i + p[i];
        }
    }
    return vector<int>(begin(p) + 1, end(p) - 1);
}
string solve(int n, string s) {
    vector<pair<int, int>> v;
    string t;
    for(int i = 0; i < n; i++)
    {
        if(s[i] == 'w')
        {
            v.push_back({i, 0});
            v.push_back({i, 1});
            t.push_back('v');
            t.push_back('v');
        }
        else 
        {
            v.push_back({i, -1});
            t.push_back(s[i]);
        }
    }
    string u;
    for(int i = 0; i < t.size(); i++)
    {
        u += t[i];
        if(i + 1 < t.size()) u += '#';
    }
    auto man = manacher_odd(u);
    vector<int> pre(n), suf(n);
    for(int i = 1; i < n; i++)
    {
        if(s[i] == 'w') pre[i] = i;
        if(s[i] == 'w' && s[i - 1] == 'w') pre[i] = pre[i - 1];
    }
    for(int i = n - 1; i >= 0; i--)
    {
        if(s[i] == 'w') suf[i] = i;
        if(s[i] == 'w' && s[i + 1] == 'w') suf[i] = suf[i + 1];
    }
    int ans = 1, ansl = 0;
    auto add = [&](int l, int r)
    {
        int len = (r - l + 1);
        if(ans < len) ans = len, ansl = l;
    };
    for(int i = 1; i < u.size() - 1; i++)
    {
        int l = i - man[i] + 1;
        int r = i + man[i] - 1;
        int L = (l + 1) / 2;
        int R = r / 2;
        if(L < R) 
        {
            auto [l, r, m1, m2] = array<int, 4>{v[L].F, v[R].F, v[L].S, v[R].S};
            if(s[l] == 'w' && s[r] == 'w' && pre[r] <= l) continue;
            if((m1 == -1 || m1 == 0) && (m2 == -1 || m2 == 1)) 
            {
                add(l, r);
                continue;
            }
            if(m1 == -1 || m2 == -1)
            {
                add(l + 1, r - 1);
                continue;
            }
            if(m1 == 0)
            {
                int d1 = suf[l] - l;
                int d2 = r - pre[r];
                int d = min(d1, d2);
                l += d;
                r -= d;
                if(s[l] == 'w' && s[r] == 'w')
                {
                    l++, r--;
                    if(s[r] == 'w') l++, r--;
                    else r--;
                }
                add(l, r);
            }
            if(m1 == 1)
            {
                int d1 = suf[l] - l;
                int d2 = r - pre[r];
                int d = min(d1, d2);
                l += d;
                r -= d;
                if(s[l] == 'w' && s[r] == 'w')
                {
                    l++, r--;
                    if(s[l] == 'w') l++, r--;
                    else l++;
                }
                add(l, r);
            }
        }
    }
    for(int i = 0; i < n; i++) if(s[i] == 'w') add(pre[i], i);
    return s.substr(ansl, ans);
}

int P(string s)
{
    string t;
    for(char c : s)
    {
        if(c == 'w') t += 'v', t += 'v';
        else t += c;
    }
    for(int i = 0; i < t.size(); i++)
    {
        if(t[i] != t[t.size() - i - 1]) return false;
    }
    return true;
}

string brute(int n, string s)
{
    int ans = 0;
    string a = "#";
    for(int i = 0; i < n; i++)
    for(int j = i; j < n; j++)
    {
        string t = s.substr(i, j - i + 1);
        if(P(t) && t.size() > ans)
        {
            ans = t.size();
            a = t;
        }
    }
    return a;
}
void test(int num, int n)
{
    while(num--)
    {
        string s(n, '#');
        for(int i = 0; i < n; i++)
        {
            int r = rand() % 3;
            if(r == 0) s[i] = 'w';
            else if(r == 1) s[i] = 'v';
            else 
            {
                s[i] = rand() % 26 + 'a';
            }
        }
        string t1 = brute(n, s);
        string t2 = solve(n, s);
        if(!P(t2))
        {
            hha(s); hha(t2);
            exit(1);
        }
        if(t1.size() != t2.size())
        {
            hha(s); hha(t1); hhal(t2);
            exit(1);
        }
    }
    exit(0);
}
