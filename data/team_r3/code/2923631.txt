void solve() {
    int n; cin >> n;
    if(n % 4 == 0) 
    {
        cout << "NO\n";
        return;
    }
    vector<int> b(32), a(32);
    for(int i = 0; i < 32; i++) b[i] = ((n >> i) & 1);
    for(int i = 31; i >= 0; i--)
    {
        if(b[i])
        {
            int k = i;
            for(int j = i + 1; j < 32; j++)
            {
                if(b[j]) break;
                k = j;
            }
            a[k] = 1;
            for(int j = i; j < k; j++) a[j] = -1;
        }
    }
    cout << "YES\n";
    for(int i = 0; i < 32; i++)
    {
        cout << a[i];
        if(i % 8 == 7) cout << "\n";
        else cout << " ";
    }
}   

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t = 1;
    cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}
