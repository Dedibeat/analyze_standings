#include<bits/stdc++.h>
using namespace std;
const int mod = 998244353;
int toi(char c)
{
    if(c >= 'a' && c <= 'z') return c - 'a';
    if(c >= 'A' && c <= 'Z') return c - 'A' + 26;
    return c - '0' + 52;
}
signed main()
{
    //ios::sync_with_stdio(0); cin.tie(0);
    int n; cin >> n;
    vector<array<int, 8>> dp(63), ndp(63);
    dp[62][0] = 1;
    for(int t = 1; t <= n; t++) {
        auto forw = [&](int j)
        {
            for(int i = 0; i < 63; i++)
            {
                for(int mask = 0; mask < 8; mask++)
                {
                    int id = 2;
                    if(j < 26) id = 0;
                    else if(j < 52) id = 1;
                    if(i != j) 
                    {
                        ndp[j][mask | (1 << id)] += dp[i][mask];
                        ndp[j][mask | (1 << id)] %= mod;
                    }
                }
            }
        };
        char c; cin >> c;
        if(c == '?')
        {
            for(int mask = 0; mask < 8; mask++)
            {
                int sum = 0;
                for(int i = 0; i < 63; i++)
                {
                    sum = (sum + dp[i][mask]) % mod;
                }
                for(int j = 0; j < 62; j++)
                {
                    int id = 2;
                    if(j < 26) id = 0;
                    else if(j < 52) id = 1;
                    int nmask = mask | (1 << id);
                    ndp[j][nmask] += sum;
                    ndp[j][nmask] %= mod;
                    ndp[j][nmask] += (mod - dp[j][mask]);
                    ndp[j][nmask] %= mod;
                }
            }
            
        }
        else 
        {
            int x = toi(c);
            if(x < 26)
            {
                forw(x);
                forw(x + 26);
            }
            else
            {
                forw(x);
            }
        }
        swap(dp, ndp);
        for(int i = 0; i < 63; i++) 
        for(int j = 0; j < 8; j++)
            ndp[i][j] = 0;
    }
    int ans = 0;
    for(int i = 0; i < 62; i++) { 
        ans = (ans + dp[i][7]) % mod;
    }
    cout << ans << '\n';
}
