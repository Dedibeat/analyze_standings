const int inf = 1e9;

int dp[105][105];

void solve() {
    int n, m;
    cin >> n >> m;
    int k;
    cin >> k;

    vector<array<int, 2>> p(k);
    vector<int> pos(m + 1, -1);
    int id = 0;

    for(auto &[r, c] : p) {
        cin >> r >> c;
        pos[c] = id;
        id++;
    }
    // print(pos);

    auto bod = [&](int mask) -> int {
        for(int j = 2; j <= m; j++) {
            for(int i = 1; i <= n; i++) {
                dp[i][j] = dp[i][j - 1] + 1;
            }
            if(pos[j] != -1)
            {
                int id = pos[j];
                if(mask >> id & 1)
                {
                    for(int i = 1; i <= p[id][0]; i++)
                        dp[i][j] = inf;
                }
                else 
                {
                    for(int i = p[id][0]; i <= n; i++)
                        dp[i][j] = inf;
                }
            }
            for(int i = 2; i <= n; i++) if(dp[i][j] != inf) dp[i][j] = min(dp[i][j], dp[i - 1][j] + 1);
            for(int i = n - 1; i >= 1; i--) if(dp[i][j] != inf) dp[i][j] = min(dp[i][j], dp[i + 1][j] + 1);
        } 
        int ans = inf;
        for(int i = 1; i <= n; i++) {
            ans = min(dp[i][m], ans);
        }
        return (ans >= (inf / 2) ? -1 : ans);
    };
    for(int i = 0; i < (1 << k); i++) {
        cout << bod(i) << " ";
    }   
    cout << '\n';
}


signed main() {
    ios_base::sync_with_stdio(0);cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
