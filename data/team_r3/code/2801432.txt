#include <bits/stdc++.h> 
using namespace std;
using ll = long long;
#define int long long
#define pb push_back
#define all(x) x.begin(), x.end()
#define rall(a) x.rbegin(), x.rend()
#define FOR(i, a, b) for(int i = (a); i < (b); i ++)
#define FORD(i, a, b) for(int i = (a); i >= (b); i--)
#define _ << ' ' <<

signed main() {
    int n; cin >> n;
    vector<int> a(n + 1); 
    FOR(i, 1, n + 1) cin >> a[i];
    vector<int> sorted_a = a;
    sort(sorted_a.begin() + 1, sorted_a.end());
    int l = 1, r = n;
    int ans = sorted_a[1];
    while(l <= r) {
        int mid = (l + r) / 2;
        int m = sorted_a[mid];
        vector<int> b(n + 1, 0);
        FOR(i, 1, n + 1) b[i] = (a[i] >= m ? 1 : -1);
        vector<int> pre(n + 1, 0);
        FOR(i, 1, n + 1) pre[i] = pre[i - 1] + b[i];
        vector<int> c(n + 1, 0);
        int sum = 0;
        FOR(r, 1, n + 1) {
            int cur = 0;
            FORD(l, r, 1) {
                if(pre[r] - pre[l - 1] > 0) cur++;
                else cur--;
                c[l] += cur;
                if(c[l] > 0) sum++;
            }
        }
        int tot = n * (n + 1) / 2;
        if(sum > tot / 2) {
            ans = m;
            l = mid + 1;
        }
        else {
            r = mid - 1;
        }
    }
    cout << ans << '\n';
}
