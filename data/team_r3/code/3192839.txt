int comp(vector<int> &a, vector<int> &b, int l)
{
    int n = a.size(), m = b.size();
    // print(a);
    // print(b);
    int mask = ((1LL << (l + 2)) - 1);
    map<int, int> mp;
    for(int i = 0; i + 1 < m; i++)
    {
        if((b[i] ^ b[i + 1]) == 1) mp[b[i] & mask]++;
    }
    int ans = 0;
    int x = (1LL << (l + 2)) - 2;
    int y = (1LL << (l + 1)) - 2;
    for(int j = 0; j + 1 < n; j++)
    {
        if((a[j] ^ a[j + 1]) == x)
        {
            // if(mp.count((a[j] & mask) ^ y))
            // {
            //     hha(l); hha(x); hhal(y); 
            //     hha(a[j]); hhal(mask);
            //     ans += mp[(a[j] & mask) ^ y];
            // }
            ans += mp[(a[j] & mask) ^ y];
        }
    }
    return ans;
}
int solve(vector<int> a, vector<int> b, int n, int m) {
    int ans = 0;
    for(int l = 0; l < 3; l++)
    {
        int mask = (1LL << (l + 1));
        int ret = comp(a, b, l);
        ans += ret;
    }
    return ans;
}
int brute(vector<int> a, vector<int> b, int n, int m)
{
    // print(a);
    // print(b);
    int ans = 0;
    for(int i = 0; i + 1 < n; i++)
    for(int j = 0; j + 1 < m; j++)
    {
        vector<int> c = {a[i] ^ b[j], a[i] ^ b[j + 1], a[i + 1] ^ b[j], a[i + 1] ^ b[j + 1]};
        // print(c);
        if(c[1] == c[0] + 1) 
        if(c[2] == c[0] + 2)
        if(c[3] == c[0] + 3)
            ans++;
    }
    return ans;
}
