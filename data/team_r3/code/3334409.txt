const int mod = 1e9 + 7;
int dp[105][105][10005];
int solve(int n, int m, vector<pair<int, int>> v) {
    v.insert(v.begin(), {1, 0});
    int ans = 1;
    vector<int> dpcur(n + 5, 0), dpnxt;
    dpcur[0] = 1;
    for(int i = 0; i < m; i++)
    {
        dpnxt.assign(n + 5, 0);
        int dlen = v[i + 1].ff - v[i].ff;
        int dval = v[i + 1].ss - v[i].ss;
        for(int j = 0; j <= v[i].ff; j++)
        {
            for(int dj = 0; dj <= dlen; dj++)
            {
                int k = j + dj;
                int dif = dval - j * dlen;
                if(dif < 0) continue;
                dpnxt[k] = (dpnxt[k] + dpcur[j] * dp[dlen][dj][dif] % mod) % mod;
            }
        }
        dpcur = dpnxt;
    }
    ans = accumulate(all(dpcur), 0LL);
    int tail = n - v[m].ff;
    while(tail--) ans = ans * 2 % mod;
    return ans;

}

{
    
    int cnt = 0;
    for(int mask = 0; mask < (1 << (n - 1)); mask++)
    {
        vector<int> a(n);
        for(int i = 0; i < n - 1; i++)
        {
            if(mask & (1 << i)) a[i + 1] = 1;
            //a[i + 1] += a[i];
        }
        //print(a);
        auto s = a;
        for(int i = 1; i < n; i++) a[i] += a[i - 1];
        for(int i = 1; i < n; i++) s[i] += s[i - 1];
        int ok = 1;
        for(auto [pos, x] : v) 
        {
            if(s[pos - 1] != x) ok = 0;
        }
        //hhal(mask);

        cnt += ok;
    }
    return cnt;
}
