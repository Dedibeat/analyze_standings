void solve() {
    int a, b, c; cin >> a >> b >> c;
    vector<vector<char>> vis((1ll << 10), vector<char>(b, 0));
    bool ok = 1;
    int a0 = a;
    while(a < c) {
        if(a % b == c % b) {
            cout << "YES\n";
            return;
        }
        int t = (a & ((1ll << 11) - 1));
        int s = (a ^ t);
        if(vis[t][s % b]) {
            break;
        }
        vis[t][s % b] = 1;
        if(ok == 0) {
            a ^= b;
        }
        else {
            a += b;
        }
        ok ^= 1;
    }
    a = a0;
    ok = 0;
    vis.assign((1ll << 10), vector<char>(b, 0));
    while(a < c) {
        if(a % b == c % b) {
            cout << "YES\n";
            return;
        }
        int t = (a & ((1ll << 11) - 1));
        int s = (a ^ t);
        if(vis[t][s % b]) {
            cout << "NO\n";
            return;
        }
        vis[t][s % b] = 1;
        if(ok == 0) {
            a ^= b;
        }
        else {
            a += b;
        }
        ok ^= 1;
    }
    cout << "NO\n";
}
signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
}
