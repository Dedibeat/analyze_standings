#include <bits/stdc++.h>
using namespace std;
#define int long long 
#define uint unsigned long long
#define all(a) (a).begin(),(a).end()
#define rall(a) (a).rbegin(),(a).rend()
#define pb push_back
#define hha(a) cout << #a << " = " << a << " ";
#define hhal(a) cout << #a << " = " << a << "\n";
#define print(x) cout << #x << " = "; for(auto a : x) cout << a << " "; cout << "\n";
const uint N = 3005, inf = (int)4e18;
int a[N], b[N];
vector<uint> dpcur, dpnxt;
int n, m, x, y, z;
uint cost(int l, int r, int i)
{
    uint tot = (uint)llabs(a[r % n] - b[i]) * y;
    assert(l < r);
    if(r >= n) tot += x;
    tot += (r - l - 1) * z;
    return tot;
}
void comp(int l, int r, int lo, int hi, int i)
{
    if(l > r) return;
    int mid = (l + r) / 2;
    pair<uint, int> best = {inf, -1};
    for(int k = max(lo, 0LL); k <= min(hi, mid - 1); k++)
    {
        best = min(best, pair<uint, int>{dpcur[k] + cost(k, mid, i), k});
    }
    dpnxt[mid] = best.first;
    int opt = best.second;
    comp(l, mid - 1, lo, opt, i);
    comp(mid + 1, r, opt, hi, i);
}
int solve() {
    dpcur.assign(2 * n, 0);
    for(int i = 0; i < n; i++)
    {
        dpcur[i] = (uint)llabs(a[i] - b[0]) * y + i * z + x;
    }
    for(int i = 1; i < m; i++)
    {
        dpnxt.assign(2 * n, inf);
        for(int i = 0; i < n; i++) dpcur[i + n] = dpcur[i];
        comp(1, 2 * n - 1, 0, 2 * n - 1, i);
        dpcur = dpnxt;
        for(int i = 0; i < n; i++) dpcur[i] = min(dpcur[i], dpcur[i + n]);
        //print(dpcur);
    }
    uint mn = inf;
    for(int i = 0; i < n; i++)
    {
        mn = min(mn, dpcur[i] + (n - i - 1) * z);
    }
    return mn;
}
int costl(int l, int r, int i)
{
    int tot = llabs(a[r] - b[i]) * y;
    if(r <= l) r += n, tot += x;
    tot += (r - l - 1) * z;
    return tot;
}
int brute()
{
    dpcur.assign(n, 0);
    for(int i = 0; i < n; i++)
    {
        dpcur[i] = llabs(a[i] - b[0]) * y + i * z + x;
    }
    for(int i = 1; i < m; i++)
    {
        dpnxt.assign(n, inf);
        for(int l = 0; l < n; l++)
        {
            for(int r = 0; r < n; r++)
            {
                dpnxt[r] = min(dpnxt[r], dpcur[l] + costl(l, r, i));
            }
        }
        dpcur = dpnxt;
        //print(dpcur);
    }
    uint mn = inf;
    for(int i = 0; i < n; i++)
    {
        mn = min(mn, dpcur[i] + (n - i - 1) * z);
    }
    return mn;
}
void test(int nn, int mm, int num, int mx)
{
    while(num--)
    {
        x = rand() % mx;
        y = rand() % mx;
        z = rand() % mx;
        n = nn;
        m = mm;
        for(int i = 0; i < n; i++) a[i] = rand() % mx;
        for(int i = 0; i < m; i++) b[i] = rand() % mx;
        int ret = solve();
        int ans = brute();
        //hha(ret); hha(ans);
        if(ret != ans)
        {
            hha(ret); hhal(ans);
            hha(n); hha(m); hha(x); hha(y); hhal(z);
            for(int i = 0; i < n; i++) cout << a[i] << " ";
            cout << endl;
            for(int i = 0; i < m; i++) cout << a[i] << " ";
            cout << endl;
            exit(0);
        }
    }
    exit(0);
}

        for(int j = 0; j < m; j++) cin >> b[j];
        cout << solve() << "\n";
    }

}
