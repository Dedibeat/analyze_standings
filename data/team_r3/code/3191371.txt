string Xor(string s, string t)
{
    if(s.size() < t.size()) swap(s, t);
    for(int i = 0; i < t.size(); i++)
    {
        int j = s.size() - t.size() + i;
        if(t[i] != s[j]) s[j] = '1';
        else s[j] = '0';
    }
    return s;
}
string Norm(string s)
{
    int i = 0;
    while(i < s.size() - 1 && s[i] == '0')
    {
        i++;
    }
    return s.substr(i, s.size() - i);
}
void solve() {
    string s; cin >> s;
    int n = s.size();
    string mx = "0";
    {
        string a = Norm(s.substr(0, 1));
        string b = Norm(s.substr(2, n - 2));
        string t = Norm(Xor(a, b));
        mx = max(mx, t);
    }
    {
        string a = Norm(s.substr(0, n - 2));
        string b = Norm(s.substr(n - 1, 1));
        string t = Norm(Xor(a, b));
        mx = max(mx, t);
    }
    cout << mx << "\n";
}
signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
}
