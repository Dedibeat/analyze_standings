void solve() {
    int l, r; cin >> l >> r;
    int cur = r;
    int ans = 0;
    while(cur > 0) {
        ans++;
        if(cur % 3) {
            cur--;
        }
        else {
            cur /= 3;
        }
    }
    cout << ans + 1 << '\n';
}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
