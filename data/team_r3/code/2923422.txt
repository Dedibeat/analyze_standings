void solve() {
    int n; cin >> n;
    vector<int> a(n + 1); FOR(i, 1, n + 1) cin >> a[i];
    int cur = 1500;
    FOR(i, 1, n + 1) {
        cur += a[i];
        if(cur >= 4000) {
            cout << i << '\n';
            return;
        }
    }
    cout << -1 << '\n';
    
}

    int t = 1;
    //cin >> t;
    while(t--) {
        solve();
    }
    return 0;
}
