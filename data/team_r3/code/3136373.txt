void solve() {
    int n, d; cin >> n >> d;
    vector<int> a1(n); for(auto &x : a1) cin >> x;
    int ans = 0;
    rep(i, 0, n) {
        vector<int> a = a1;
        int l = a[i];
        int r = l + d;
        rep(j, 0, n) {
            if(a[j] < l) a[j] = l;
            else if(a[j] > r) a[j] = r;
        }
        int cur = 0;
        rep(j, 1, n) {
            cur += abs(a[j] - a[j - 1]);
        }
        ans = max(ans, cur);
        a = a1;
        r = a[i];
        l = r - d;
        rep(j, 0, n) {
            if(a[j] < l) a[j] = l;
            else if(a[j] > r) a[j] = r;
        }
        cur = 0;
        rep(j, 1, n) {
            cur += abs(a[j] - a[j - 1]);
        }
        ans = max(ans, cur);
    }
    cout << ans << '\n';
}
signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    //int t; cin >> t; while(t--);
    solve();
}
