#include<bits/stdc++.h>
using namespace std;
#define pb push_back
vector<vector<pair<int, int>>> g;
vector<char> vis;
vector<char> edgeVis;
signed main()
{
    ios::sync_with_stdio(0); cin.tie(0);
    int n, m; cin >> n >> m;
    g.resize(n + 1);
    vis.assign(n + 1, 0);
    edgeVis.assign(m + 1, 0);
    for(int i = 1; i <= m; i ++) {
        int u, v; cin >> u >> v;
        g[u].pb({v, i});
        g[v].pb({u, i});
    }
    long long mn1 = LLONG_MAX, mn2 = LLONG_MAX;
    long long ans = 0;
    for(int i = 1; i <= n; i++) {
        if(vis[i] == 1) continue;
        vis[i] = 1;
        int cnt = 1;
        int cnt1 = 0;
        queue<int> q; 
        q.push(i);
        while(!q.empty()) {
            int u = q.front(); q.pop();
            for(auto &[v, i] : g[u]) {
                if(edgeVis[i] == 0) {
                    edgeVis[i] = 1;
                    cnt1++;
                }
                if(vis[v]) continue;
                cnt++;
                q.push(v);
                vis[v] = 1;
            }
        }
        if(1ll * cnt * (cnt - 1) / 2 > 1ll * cnt1) {
            ans += 1ll * cnt * (cnt - 1) / 2 - 1ll * cnt1;
        }
        if(cnt < mn1) {
            mn2 = mn1;
            mn1 = cnt;
        }
        else if(cnt < mn2) {
            mn2 = cnt;
        }
    }
    if(ans) {
        cout << ans << '\n';
    }
    else {
        cout << 1ll * mn1 * mn2 << '\n';
    }
}
