void solve() {
    int n, c1, c2; cin >> n >> c1 >> c2;
    int ans = 0;
    rep(i, 0, n) {
        set<char> st;
        rep(j, 0, 3) {
            char c; cin >> c;
            st.insert(c);
        }
        if(st.size() == 3) {
            ans += min(c1, c2) * 3;
        }
        else {
            ans += min(min(c1, c2) * 3, c2 + min(c1, c2));
        }
    }
    cout << ans << '\n';
}

    //int t; cin >> t; while(t--)
    solve();
