struct TreeCanon {
    map<vector<int>,int> id;
    int get(vector<int> children) {
        sort(children.begin(),children.end());
        auto [it,inserted]=id.emplace(children,id.size());
        return it->second;
    }
    int rootedDfs(const vector<vector<int>>& g,int v,int p, vector<int> &vis) {
        vector<int>children;
        for(int u:g[v]) if(u!=p && !vis[u]) children.push_back(rootedDfs(g,u,v,vis));
        return get(children);
    }
};
void solve() {
    TreeCanon canon;
    int n, m; cin >> n >> m;
    vector<vector<int>> g(n);
    for(int i = 0; i < m; i++)
    {
        int u, v; cin >> u >> v;
        u--, v--;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    if(m > n)
    {
        cout << "NO\n";
        return;
    }
    if(m == n - 1)
    {
        cout << "YES\n";
        return;
    }
    vector<int> par(n), vis(n);
    vector<int> cyc; 
    int x = -1, y = -1;
    auto findcyc = [&](auto &&self, int u, int p) -> void
    {
        vis[u] = true;
        for(int v : g[u])
        {
            if(v == p) continue;
            if(vis[v])
            {
                if(x < 0 && y < 0)
                {
                    x = v; y = u;
                }
            }
            else 
            {
                par[v] = u;
                self(self, v, u);
            }
        }
    };
    findcyc(findcyc, 0, 0);
    //hha(x); hhal(y);
    int cnt = 1e18;
    while(y != x && cnt--) 
    {
        cyc.push_back(y);
        y = par[y];
    }
    //print(par);
    cyc.push_back(x);
    //print(cyc);
    fill(all(vis), 0);
    for(int u : cyc) vis[u] = true;
    vector<int> a;
    for(int u : cyc) 
    {
        a.push_back(canon.rootedDfs(g, u, -1, vis));
    }
    int sz = a.size();
    int ok = 1;
    if(sz % 2 == 0)
    {
        for(int i = 2; i < sz; i++)
        {
            if(a[i - 2] != a[i]) ok = 0;
        }
    }
    else 
    {
        for(int i = 1; i < sz; i++)
        {
            if(a[i - 1] != a[i]) ok = 0;
        }
    }
    cout << (ok ? "YES\n" : "NO\n");
    // print(a);
}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
