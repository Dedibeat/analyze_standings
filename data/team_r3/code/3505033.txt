const int N = 2e5 + 5;
const int mod = 998244353;
int fact[N], invfact[N], inv[N];
void solve()
{
    int n; cin >> n;
    vector<int> p(n + 1), a(n + 1), b(n + 1);
    for(int i = 2; i <= n; i++) cin >> p[i];
    for(int i = 1; i <= n; i++) cin >> a[i];
    for(int i = 1; i <= n; i++) cin >> b[i];
    vector<vector<int>> g(n + 1);
    vector<int> par(n + 1);
    for(int i = 2; i <= n; i++){
        if(b[i] > a[p[i]] || a[i] > b[p[i]])
        {
            par[i] = p[i];
        }
    }
    for(int i = 1; i <= n; i++)
    {
        g[par[i]].push_back(i);
    }
    vector<int> dp(n + 1), sz(n + 1);
    auto dfs = [&](auto &&self, int u) -> void
    {
        sz[u] = 1;
        int tot = 1;
        for(int v : g[u])
        {
            self(self, v);
            sz[u] += sz[v];
            tot = tot * dp[v] % mod;
            tot = tot * invfact[sz[v]] % mod;
        }
        tot = tot * fact[sz[u] - 1] % mod;
        dp[u] = tot;
    };
    dfs(dfs, 0);
    cout << dp[0] << "\n";


}
signed main()
{
    ios::sync_with_stdio(0); cin.tie(0); 
    fact[0] = invfact[0] = fact[1] = invfact[1] = 1;
    inv[1] = 1;
    for(int i = 2; i < N; i++)
    {
        fact[i] = fact[i - 1] * i % mod;
        inv[i] = mod - mod / i * inv[mod % i] % mod;
        invfact[i] = invfact[i - 1] * inv[i] % mod;
    }
    int t; cin >> t; while(t--)
    solve();
}
