void solve() {
    int n; cin >> n;
    vector<int> deg(n + 1, 0);
    vector<int> par(n + 1, 0);
    rep(i, 2, n + 1) {
        int x; cin >> x;
        par[i] = x;
        deg[x]++;
    }
    set<int> leaf;
    rep(i, 1, n + 1) {
        if(deg[i] == 0) {
            leaf.insert(i);
        }
    }
    int ans = n;
    int cur = 0;
    while(leaf.size()) {
        ans = min(ans, cur + (int)leaf.size());
        set<int> st;
        for(auto &v : leaf) {
            deg[par[v]]--;
            if(deg[par[v]] == 0) {
                st.insert(par[v]);
            }
        }
        cur++;
        leaf.swap(st);
    }
    ans = min(ans, cur);
    cout << ans << '\n';
}

signed main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    int t; cin >> t; while(t--)
    solve();
    return 0;
}
